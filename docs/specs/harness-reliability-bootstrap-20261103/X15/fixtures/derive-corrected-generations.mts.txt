import { createHash } from "node:crypto";
import { readFileSync, writeFileSync, mkdirSync } from "node:fs";
import { execFileSync } from "node:child_process";
import { canonicalHash } from "./src/bootstrap/resultGate.ts";

const R = "docs/specs/harness-reliability-bootstrap-20261103";
const TASK = "2eee4d1a9592b839f0dd3eb677dd9d7d8ed8d59b"; // X15 task commit (scanner fix)
const EVID = "9790f21c254d788cc830d047a37781336cfba4f3"; // X15 evidence commit
const TRUTH = `${R}/X15/logs/04-link-truth-table.txt`;
const read = (p) => JSON.parse(readFileSync(p, "utf8"));
const sha = (o) => "sha256:" + createHash("sha256").update(JSON.stringify(o)).digest("hex");
const patchOf = (path, base) =>
  "sha256:" +
  createHash("sha256")
    .update(execFileSync("git", ["diff", `${base}..${TASK}`, "--", path], { encoding: "utf8", maxBuffer: 1 << 28 }))
    .digest("hex");

const correction =
  "X15 correction: the forward pointer carried by this generation names a record that was never emitted, and the" +
  " unit that actually continued from this commit is named instead. The correction is a new generation" +
  " (out-chain); the record as first emitted stays in `out` unchanged, and AUDIT/integrity-scan.mjs reads" +
  " generations from highest rank down, so the corrected pointer is canonical and the original is reported as a" +
  " superseded-generation link. Every field not named below is byte-identical to the record it copies; see" +
  " X15/logs/04-link-truth-table.txt for the commit facts, and X15/logs/01 and 03 for the scanner defect that" +
  " had hidden part of it (unit ids ending in a letter were skipped outright, and the script had no exit status).";

const PLAN = [

const out = [];
for (const p of PLAN) {
  const d = read(p.src);
  const child = read(p.child);
  d.nextTaskId = p.nextTaskId;
  d.nextBaseSha = child.baseSha;
  d.evidence = [...new Set([...d.evidence, TRUTH])];
  d.operatorApproval = {
    ...d.operatorApproval,
    notes: `${d.operatorApproval.notes}\n\n${correction}\n\nBasis: ${p.why}`,
  };
  delete d.handoffHash;
  d.handoffHash = canonicalHash(d);
  const dir = `${R}/${p.unit}/out-chain`;
  mkdirSync(dir, { recursive: true });
  writeFileSync(`${dir}/${p.unit}.handoff.json`, JSON.stringify(d, null, 2) + "\n");
  out.push({
    unit: p.unit,
    copied: p.src,
    next: `${d.nextTaskId}@${d.nextBaseSha.slice(0, 12)}`,
    childBaseMatchesChildRecord: d.nextBaseSha === child.baseSha,
    hash: d.handoffHash,
    verify: canonicalHash(d) === d.handoffHash,
    evidenceCount: d.evidence.length,
    notesBytes: d.operatorApproval.notes.length,
  });
}
